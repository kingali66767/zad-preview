أَعُوذُ بِاللَّهِ مِنَ الشَّيطَانِ الرَّجِيمِ ﴿اللَّهُ لاَ إِلَهَ إِلاَّ هُوَ الْحَيُّ الْقَيُّومُ لاَ تَأْخُذُهُ سِنَةٌ وَلاَ نَوْمٌ لَّهُ مَا فِي السَّمَوَاتِ وَمَا فِي الأَرْضِ مَن ذَا الَّذِي يَشْفَعُ عِنْدَهُ إِلاَّ بِإِذْنِهِ يَعْلَمُ مَا بَيْنَ أَيْدِيهِمْ وَمَا خَلْفَهُمْ وَلاَ يُحِيطُونَ بِشَيْءٍ مِّنْ عِلْمِهِ إِلاَّ بِمَا شَاء وَسِعَ كُرْسِيُّهُ السَّمَوَاتِ وَالْأَرْضَ وَلاَ يَؤُودُهُ حِفْظُهُمَا وَهُوَ الْعَلِيُّ الْعَظِيمُ﴾

مفہوم صرف اللہ عبادت کے لائق ہے۔ وہ ہمیشہ زندہ اور سب کا قائم رکھنے والا ہے۔ آسمانوں اور زمین کی نگہبانی میں اسے تھکن یا نیند نہیں آتی۔

ہر سورت ۳ بار

قُلْ هُوَ اللَّهُ أَحَدٌ ۝ اللَّهُ الصَّمَدُ ۝ لَمْ يَلِدْ وَلَمْ يُولَدْ ۝ وَلَمْ يَكُنْ لَهُ كُفُوًا أَحَدٌ
قُلْ أَعُوذُ بِرَبِّ الْفَلَقِ ۝ مِنْ شَرِّ مَا خَلَقَ ۝ وَمِنْ شَرِّ غَاسِقٍ إِذَا وَقَبَ ۝ وَمِنْ شَرِّ النَّفَّاثَاتِ فِي الْعُقَدِ ۝ وَمِنْ شَرِّ حَاسِدٍ إِذَا حَسَدَ
قُلْ أَعُوذُ بِرَبِّ النَّاسِ ۝ مَلِكِ النَّاسِ ۝ إِلَهِ النَّاسِ ۝ مِنْ شَرِّ الْوَسْوَاسِ الْخَنَّاسِ ۝ الَّذِي يُوَسْوِسُ فِي صُدُورِ النَّاسِ ۝ مِنَ الْجِنَّةِ وَالنَّاسِ

مفہوم کہہ دیجیے کہ اللہ ایک ہے۔ اللہ بے نیاز ہے، سب اس کے محتاج ہیں۔ نہ اس کی کوئی اولاد ہے اور نہ وہ کسی کی اولاد ہے۔ کوئی اس کے برابر نہیں۔ کہہ دیجیے کہ میں صبح کے رب کی پناہ مانگتا ہوں، اس کی مخلوق کے شر سے، اندھیری رات کے شر سے جب وہ چھا جائے، گرہوں میں پھونکنے والیوں کے شر سے، اور حسد کرنے والے کے شر سے جب وہ حسد کرے۔ کہہ دیجیے کہ میں لوگوں کے رب، لوگوں کے بادشاہ، لوگوں کے معبود کی پناہ مانگتا ہوں، اس وسوسہ ڈالنے والے کے شر سے جو پیچھے ہٹ جاتا ہے، جو لوگوں کے دلوں میں وسوسہ ڈالتا ہے، خواہ وہ جنوں میں سے ہو یا انسانوں میں سے۔

تینوں سورتیں صبح اور شام پڑھی جاتی ہیں۔ یہ تعداد عبداللہ بن خبیب رضی اللہ عنہ کی حدیث میں آئی ہے۔

بادشاہی اللہ کے لیے ہے، صبح کی دعا

أَصْبَحْنَا وَأَصْبَحَ الْمُلْكُ لِلَّهِ، وَلَا إِلَهَ إِلَّا اللَّهُ وَحْدَهُ لَا شَرِيكَ لَهُ، لَهُ الْمُلْكُ وَلَهُ الْحَمْدُ وَهُوَ عَلَى كُلِّ شَيْءٍ قَدِيرٌ. اللَّهُمَّ إِنِّي أَسْأَلُكَ خَيْرَ هَذَا الْيَوْمِ وَخَيْرَ مَا فِيهِ وَخَيْرَ مَا بَعْدَهُ، وَأَعُوذُ بِكَ مِنْ شَرِّ هَذَا الْيَوْمِ وَشَرِّ مَا فِيهِ وَشَرِّ مَا بَعْدَهُ. رَبِّ أَعُوذُ بِكَ مِنَ الْكَسَلِ وَسُوءِ الْكِبَرِ، وَأَعُوذُ بِكَ مِنْ عَذَابٍ فِي النَّارِ وَعَذَابٍ فِي الْقَبْرِ.

مفہوم ہم نے صبح کی اور بادشاہی اللہ کی ہے۔ اللہ کے سوا کوئی معبود برحق نہیں۔ وہ اکیلا ہے، اس کا کوئی شریک نہیں۔ اسی کی بادشاہی ہے اور اسی کے لیے حمد ہے۔ وہ ہر چیز پر قادر ہے۔ اے اللہ! میں تجھ سے اس دن کی خیر، اس میں موجود خیر اور اس کے بعد کی خیر مانگتا ہوں۔ اور اس دن کے شر، اس کے اندر کے شر اور اس کے بعد کے شر سے تیری پناہ مانگتا ہوں۔ میرے رب! میں سستی اور بڑھاپے کی بدحالی سے تیری پناہ مانگتا ہوں، اور آگ کے عذاب اور قبر کے عذاب سے بھی تیری پناہ مانگتا ہوں۔

یہ صبح کے الفاظ ہیں جو شیخ ابن باز نے اپنے فتوے میں ذکر کیے ہیں۔ ابن مسعود رضی اللہ عنہ کی حدیث میں بھی صبح کے اس ذکر کا آغاز بیان ہوا ہے۔

اے اللہ! تیرے ہی حکم سے ہم نے صبح کی

اللَّهُمَّ بِكَ أَصْبَحْنَا، وَبِكَ أَمْسَيْنَا، وَبِكَ نَحْيَا، وَبِكَ نَمُوتُ، وَإِلَيْكَ النُّشُورُ.

مفہوم اے اللہ! تیرے ہی حکم سے ہم نے صبح کی، تیرے ہی حکم سے شام کی، تیرے ہی حکم سے ہم جیتے ہیں اور تیرے ہی حکم سے مرتے ہیں، اور تیری ہی طرف دوبارہ اٹھنا ہے۔

ابو ہریرہ رضی اللہ عنہ کی حدیث میں صبح کی یہ دعا آئی ہے۔

صحیح احادیث 1199

سید الاستغفار

اللَّهُمَّ أَنْتَ رَبِّي لَا إِلَهَ إِلَّا أَنْتَ، خَلَقْتَنِي وَأَنَا عَبْدُكَ، وَأَنَا عَلَى عَهْدِكَ وَوَعْدِكَ مَا اسْتَطَعْتُ، أَعُوذُ بِكَ مِنْ شَرِّ مَا صَنَعْتُ، أَبُوءُ لَكَ بِنِعْمَتِكَ عَلَيَّ، وَأَبُوءُ لَكَ بِذَنْبِي، فَاغْفِرْ لِي، فَإِنَّهُ لَا يَغْفِرُ الذُّنُوبَ إِلَّا أَنْتَ.

مفہوم اے اللہ! تو میرا رب ہے، تیرے سوا کوئی معبود برحق نہیں۔ تو نے مجھے پیدا کیا اور میں تیرا بندہ ہوں۔ جہاں تک میری طاقت ہے میں تیرے عہد اور وعدے پر قائم ہوں۔ میں اپنے کیے ہوئے کاموں کے شر سے تیری پناہ مانگتا ہوں۔ میں اپنے اوپر تیری نعمتوں کا اعتراف کرتا ہوں اور اپنے گناہ کا بھی اعتراف کرتا ہوں۔ مجھے بخش دے، کیونکہ تیرے سوا کوئی گناہ نہیں بخش سکتا۔

شداد بن اوس رضی اللہ عنہ کی حدیث میں نبی ﷺ نے یقین کے ساتھ دن اور رات میں یہ دعا پڑھنے کی فضیلت بیان کی ہے۔

صحیح بخاری 6306

عافیت اور حفاظت کی دعا

تین مرتبہ

اللَّهُمَّ عَافِنِي فِي بَدَنِي، اللَّهُمَّ عَافِنِي فِي سَمْعِي، اللَّهُمَّ عَافِنِي فِي بَصَرِي، لاَ إِلَهَ إِلاَّ أَنْتَ. اللَّهُمَّ إِنِّي أَعُوذُ بِكَ مِنَ الْكُفْرِ، وَالفَقْرِ، وَأَعُوذُ بِكَ مِنْ عَذَابِ القَبْرِ، لاَ إِلَهَ إِلاَّ أَنْتَ

مفہوم اللہ سے بدن، سماعت اور بصارت کی عافیت، اور کفر، محتاجی اور عذابِ قبر سے پناہ مانگی جاتی ہے۔

حَسْبِيَ اللَّهُ لَا إِلَٰهَ إِلَّا هُوَ، عَلَيْهِ تَوَكَّلْتُ، وَهُوَ رَبُّ الْعَرْشِ الْعَظِيمِ.

مفہوم میرے لیے اللہ کافی ہے اور عبادت کا مستحق صرف وہی ہے۔ میرا بھروسا اسی پر ہے جو عرشِ عظیم کا رب ہے۔

معافی اور عافیت کی دعا

اللَّهُمَّ إِنِّي أَسْأَلُكَ الْعَافِيَةَ فِي الدُّنْيَا وَالْآخِرَةِ، اللَّهُمَّ إِنِّي أَسْأَلُكَ الْعَفْوَ وَالْعَافِيَةَ فِي دِينِي وَدُنْيَايَ وَأَهْلِي وَمَالِي، اللَّهُمَّ اسْتُرْ عَوْرَاتِي وَآمِنْ رَوْعَاتِي، اللَّهُمَّ احْفَظْنِي مِنْ بَيْنِ يَدَيَّ وَمِنْ خَلْفِي وَعَنْ يَمِينِي وَعَنْ شِمَالِي وَمِنْ فَوْقِي، وَأَعُوذُ بِعَظَمَتِكَ أَنْ أُغْتَالَ مِنْ تَحْتِي.

مفہوم اے اللہ! میں تجھ سے دنیا اور آخرت میں عافیت مانگتا ہوں۔ اے اللہ! میں تجھ سے اپنے دین، اپنی دنیا، اپنے گھر والوں اور اپنے مال میں معافی اور عافیت مانگتا ہوں۔ اے اللہ! میری پردہ پوشی فرما اور مجھے میرے خوف سے امن عطا فرما۔ اے اللہ! میرے آگے، میرے پیچھے، میرے دائیں، میرے بائیں اور میرے اوپر سے میری حفاظت فرما۔ اور میں تیری عظمت کی پناہ مانگتا ہوں کہ نیچے سے اچانک ہلاک کر دیا جاؤں۔

عبداللہ بن عمر رضی اللہ عنہما کی حدیث میں یہ ان دعاؤں میں سے ہے جن پر نبی ﷺ صبح و شام پابندی کرتے تھے۔

سنن ابی داؤد 5074

نفس اور شیطان کے شر سے پناہ

اللَّهُمَّ فَاطِرَ السَّمَاوَاتِ وَالْأَرْضِ، عَالِمَ الْغَيْبِ وَالشَّهَادَةِ، رَبَّ كُلِّ شَيْءٍ وَمَلِيكَهُ، أَشْهَدُ أَنْ لَا إِلَهَ إِلَّا أَنْتَ، أَعُوذُ بِكَ مِنْ شَرِّ نَفْسِي وَشَرِّ الشَّيْطَانِ وَشِرْكِهِ.

مفہوم اے اللہ! آسمانوں اور زمین کو پیدا کرنے والے، غیب اور ظاہر کے جاننے والے، ہر چیز کے رب اور مالک! میں گواہی دیتا ہوں کہ تیرے سوا کوئی معبود برحق نہیں۔ میں اپنے نفس کے شر اور شیطان کے شر اور اس کے شرک سے تیری پناہ مانگتا ہوں۔

نبی ﷺ نے یہ دعا ابو بکر صدیق رضی اللہ عنہ کو سکھائی اور صبح، شام اور بستر پر جاتے وقت اسے پڑھنے کی ہدایت دی۔

اللہ کے نام کے ساتھ حفاظت کی دعا

مفہوم اللہ کے نام سے، جس کے نام کے ساتھ زمین یا آسمان میں کوئی چیز نقصان نہیں پہنچا سکتی، اور وہ سب کچھ سننے والا، سب کچھ جاننے والا ہے۔

أَصْبَحْنَا عَلَى فِطْرَةِ الْإِسْلَامِ، وَكَلِمَةِ الْإِخْلَاصِ، وَدِينِ نَبِيِّنَا مُحَمَّدٍ ﷺ، وَمِلَّةِ أَبِينَا إِبْرَاهِيمَ حَنِيفًا مُسْلِمًا، وَمَا كَانَ مِنَ الْمُشْرِكِينَ.

مفہوم ہم نے اسلام کی فطرت، کلمۂ اخلاص، اپنے نبی محمد ﷺ کے دین اور اپنے باپ ابراہیم کے طریقے پر صبح کی۔ وہ اللہ کی طرف یکسو مسلمان تھے اور مشرکوں میں سے نہیں تھے۔

مفہوم اللہ پاک ہے اور میں اس کی حمد کرتا ہوں۔

ابو ہریرہ رضی اللہ عنہ کی حدیث کے مطابق یہ ذکر صبح سو بار اور شام سو بار کہا جاتا ہے۔

صحیح مسلم 2692

صبح و شام کا کلمۂ توحید

لَا إِلَهَ إِلَّا اللَّهُ وَحْدَهُ لَا شَرِيكَ لَهُ، لَهُ الْمُلْكُ وَلَهُ الْحَمْدُ، وَهُوَ عَلَى كُلِّ شَيْءٍ قَدِيرٌ.

مفہوم اللہ کے سوا کوئی عبادت کے لائق نہیں۔ بادشاہی اور حمد اسی کے لیے ہیں اور وہ ہر چیز پر قادر ہے۔

یہ ابو عیاش کی صبح و شام والی روایت کے الفاظ ہیں۔ اس روایت میں کوئی تعداد مقرر نہیں ہے۔

صحیح مسلم 2726a

مفید علم، پاک رزق اور مقبول عمل

اللَّهُمَّ إِنِّي أَسْأَلُكَ عِلْماً نَافِعاً، وَرِزْقاً طَيِّباً، وَعَمَلاً مُتَقَبَّلاً

مفہوم اللہ سے فائدہ دینے والا علم، پاک رزق اور قبول ہونے والا عمل مانگا جاتا ہے۔
